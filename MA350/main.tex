\subsection{Introduction}
Motivating question: consider a triangle. How many symmetries are there?

The answer turns out to be $6$ through rotations and reflections, just the permutations of the vertices. We call this set $D_6$, with composition as the operation. Together, this makes a \vocab{group}.
Note that all elements of $D_6$ can be written as compositions of rotations and reflections. We can then say that rotation and reflection \vocab{generate} the group $D_6$.
\begin{definition}
  A \vocab{group} is a set $G$ equipped with a binary operation $\cdot: G \times G \to G$ satisfying the following axioms:
  \begin{enumerate}
  \item Associativity: $(a \cdot b) \cdot c = a \cdot (b \cdot c)$
  \item Identity: $\exists 1 \in G$ s.t. $a \cdot 1 = a$.
  \item Inverse: $\forall a \in G$, $\exists a^{-1}$ s.t. $a^{-1} \cdot a = 1$
  \item Closed: $\forall a,b \in G$, $a \cdot b \in G$. (This is kind of implicit in our definition of $\cdot$).
  \end{enumerate}
\end{definition}
Some other groups/rings:
\begin{itemize}
\item $(\ZZ, +)$ 
\item $(\QQ \setminus 0, \times)$
\item $(\ZZ, +, \times)$: Ring
\item $(\ZZ/12, +)$
\item $S_3$: all the permutations of $3$ elements
\end{itemize}

\begin{theorem}{First Isomorphism Theorem}{}
Let $G$ be a group and $\phi: G \to H$ be a homomorphism. 
Then $\ker \phi \leq G$ is a normal subgroup and $G / \ker \phi \cong \phi(G) := \Im \phi$.
\end{theorem}
In general, if we suspect $G / K \cong H$, then we try to write a map $G \to H$ with kernel $K$.
\begin{definition}
  Let $A_n$ (alternating group on $n$ letters) be the kernel of $\sgn: S_n \to (\{\pm 1\}, \times)$.
\end{definition}
Then $S_n / A_n \cong \ZZ_2$. 
We want to find normal subgroups $K \triangleleft G$ and study $K$ and $G/K$ in place of $G$.
\begin{definition}
  A group $G$ is \vocab{simple} when its only normal subgroups are $\{e\}$ and $G$ itself.
\end{definition}
Then, for example, $A_5$ and $\ZZ_p$ are simple groups.

\begin{theorem}{Second Isomorphism Theorem}{}
Let $G$ be a group with subgroups $A$ and $B$. Suppose $A \leq N_G(B)$. Then $AB \leq G$, and $B \trianglelefteq AB$, and $A \cap B \trianglelefteq A$, and we have an isomorphism $AB/B \simeq A / A \cap B$. 
\end{theorem}
\proof $AB \leq N_G(B)$ implies $AB$ is a group, $B \trianglelefteq AB$, and $A \cap B \trianglelefteq A$. This is because we can write $abab = a(ab)b$. Also, $B$ is normalized by $B$ and by $A$. So $B$ is normalized by $AB$. Also if $x \in A \cap B$, then $a^{-1}xa \in B$ since $x\in B$, $a \in N_G(B)$. And we also automatically have $a^{-1}xa \in A$ so $a^{-1}xa \in A \cap B$.

Now we construct the composition $\phi: A \to AB/B$ as $A \to AB \to AB/B$. By the first isomorphism theorem, we have $A/\ker{\phi} \simeq AB/B$. Suppose $\phi(a) = e$ for $a \in A$. We wish to show that $a \in B$. This means $aB=B$ as cosets $\iff a \in B$. Thus $A/A\cap B \simeq AB/B$.

\begin{theorem}{Third Isomorphism Theorem}{}
Let $G$ be a group, and $H$ and $K$ normal subgroups of $H$ with $H \leq K$. Then $K / H \trianglelefteq G/H$ and $(G/H)/(K/H) \simeq G/K$. 
\end{theorem}
\proof
Convince yourself that $K/H \trianglelefteq G/H$. We define $\phi: G/H \to G/K$ by $gH \mapsto gK$. Then $\ker \phi = \{gH \in G/H \mid \phi(gH) = 1K\} = \{gH \in G/H \mid gK=K\} = \{gH \in G/H \mid g \in K\} \simeq K/H$.

\begin{theorem}
  Let $G$ be a group and $N \trianglelefteq G$. There's a bijection from subgroups $A \leq G$ containing $N$ to $\bar{A} = A/N$ of subgroups of $G/N$. Furthermore, for any $A$, $B$ on the LHS:
  \begin{itemize}
  \item $A \leq B \iff \bar{A} \leq \bar{B}$ 
  \item If $A \leq B$, $[B:A] = [\bar{B}:\bar{A}]$
  \item $\overline{\angled{A,B}} = \angled{\bar{A},\bar{B}}$
  \item $\overline{A \cap B} = \bar{A} \cap \bar{B}$
  \item $A \trianglelefteq B \iff \bar{A} \trianglelefteq G/N$
  \end{itemize}
\end{theorem}
Recall $[B:A]$ is the number of $A$-cosets in $B$.
\proof
Let $\phi: G \to G/N$. Given any subgroup $\bar{A} \leq G/N$, $\phi^{-1}(\bar{A})$ is a subgroup. Given any subgroup $A \leq G$ containing $N$, $\phi(A) \leq G/N$.
Note that $\bar{A}_1 \leq \bar{A}_2 \iff \phi^{-1}(\bar{A}_1) \leq \phi^{-1}(\bar{A}_2)$.

\begin{prop}
Suppose $G$ is a finite abelian group and $p$ is a prime with $p \mid \abs{G}$. Then $G$ has an element of order $p$. 
\end{prop}

\begin{theorem}{Jordan-Holder}{}
Every finite group admits a composition series, and there is a permutation $\pi \in S_r$ such that the quotients are the same for any composition series up to isomorphism. 
\end{theorem}

\begin{theorem}
If $H \leq G$ with $[G:H] = 2$, $H$ is normal. 
\end{theorem}

\begin{theorem}
  Suppose $p$ is the smallest prime dividing $\abs{G}$. Then any index $p$ subgroup is normal. That is, $[G:H]=p \implies H \trianglelefteq G$. 
\end{theorem}
